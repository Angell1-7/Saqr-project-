    <script>
        const $ = id => document.getElementById(id);
        const fileInput = $('audioFile'), dropArea = $('dropArea');
        const MAX_MB = 20;

        /* ===== بيانات الحالات الثلاث ===== */
        const OUTCOMES = {
            real: {
                label: 'صوت حقيقي (Authentic)', risk: 'منخفض', color: 'var(--ok)', confidence: 98.4,
                sub: 'ما لقينا علامات توليد بالذكاء الاصطناعي',
                desc: 'اهتزازات طبيعية وتفاوت واضح بين الترددات',
                tips: ['الصوت يبدو طبيعي، لكن لا تشارك رموز التحقق عبر الهاتف أبداً',
                       'إذا طلب منك أحد تحويل مبلغ، تأكد منه باتصال على رقمه المعروف']
            },
            fake: {
                label: 'صوت مزيف (AI Generated)', risk: 'عالي', color: 'var(--bad)', confidence: 95.2,
                sub: 'تحذير! الصوت مولّد بالذكاء الاصطناعي',
                desc: 'خطوط ناعمة ومثالية بزيادة، وهذا نمط الأصوات المولدة',
                tips: ['أنهِ المكالمة فوراً ولا تعطي أي بيانات',
                       'لا تشارك رمز التحقق (OTP) ولا رقم البطاقة مع أي أحد',
                       'اتصل بالبنك على الرقم الرسمي المكتوب خلف البطاقة',
                       'بلّغ عن محاولة الاحتيال للجهات المختصة']
            },
            warn: {
                label: 'مشكوك فيه (Suspicious)', risk: 'متوسط', color: 'var(--warn)', confidence: 62.7,
                sub: 'فيه مؤشرات تحتاج حذر والنتيجة غير حاسمة',
                desc: 'جزء من المقطع طبيعي وجزء ناعم بشكل غير معتاد',
                tips: ['لا تتصرف بناءً على هذا المقطع وحده',
                       'تأكد من الشخص عبر قناة ثانية قبل أي إجراء',
                       'جرّب مقطع أوضح وبدون ضوضاء إذا أمكن']
            }
        };
        let current = 'real', currentFile = null;

        /* ===== اختيار الملف + التحقق ===== */
        $('pickBtn').addEventListener('click', () => fileInput.click());

        function setError(msg) { $('errorMsg').textContent = msg || ''; }

        function handleFile(file) {
            setError('');
            $('resultsSection').style.display = 'none';
            if (!file) return;
            if (!file.type.startsWith('audio/')) {
                setError('الملف لازم يكون صوتي (mp3, wav, m4a, ogg).');
                $('analyzeBtn').style.display = 'none';
                return;
            }
            if (file.size > MAX_MB * 1024 * 1024) {
                setError('حجم الملف كبير، الحد الأقصى ' + MAX_MB + ' ميجا.');
                $('analyzeBtn').style.display = 'none';
                return;
            }
            currentFile = file;
            $('fileNameDisplay').textContent = 'الملف المختار: ' + file.name;
            $('analyzeBtn').style.display = 'inline-block';
        }

        fileInput.addEventListener('change', () => handleFile(fileInput.files[0]));

        /* ===== السحب والإفلات ===== */
        ['dragenter', 'dragover'].forEach(ev => dropArea.addEventListener(ev, e => {
            e.preventDefault(); dropArea.classList.add('dragover');
        }));
        ['dragleave', 'drop'].forEach(ev => dropArea.addEventListener(ev, e => {
            e.preventDefault(); dropArea.classList.remove('dragover');
        }));
        dropArea.addEventListener('drop', e => handleFile(e.dataTransfer.files[0]));

        /* ===== التحليل ===== */
        $('analyzeBtn').addEventListener('click', async () => {
            const btn = $('analyzeBtn');
            btn.disabled = true;
            btn.textContent = 'جاري التحليل...';
            $('resultsSection').style.display = 'none';
            const prog = $('progress'), bar = $('progressBar');
            prog.style.display = 'block'; bar.style.width = '0';
            let p = 0;
            const t = setInterval(() => { p = Math.min(p + 8, 90); bar.style.width = p + '%'; }, 120);

            try {
                const data = await analyzeAudio(currentFile);
                bar.style.width = '100%';
                showResults(data);
            } catch (err) {
                setError('صار خطأ أثناء التحليل، حاول مرة ثانية.');
            } finally {
                clearInterval(t);
                setTimeout(() => { prog.style.display = 'none'; }, 300);
                btn.disabled = false;
                btn.textContent = 'بدء التحليل الفوري';
            }
        });

        /* هذي الدالة هي اللي تبدلونها لاحقاً بالربط مع الـ Backend (Python) */
        async function analyzeAudio(file) {
            // ---- الربط الحقيقي (فعّلوه لما يجهز الموديل) ----
            // const fd = new FormData();
            // fd.append('audio', file);
            // const res = await fetch('/analyze', { method: 'POST', body: fd });
            // return await res.json(); // { result: 'real'|'fake'|'warn', confidence: 97.5 }

            // ---- محاكاة للعرض ----
            await new Promise(r => setTimeout(r, 1500));
            let mode = $('demoMode').value;
            if (mode === 'auto') mode = ['real', 'fake', 'warn'][file.size % 3];
            const jitter = (Math.random() * 3 - 1.5);
            return { result: mode, confidence: +(OUTCOMES[mode].confidence + jitter).toFixed(1) };
        }

        /* ===== عرض النتائج ===== */
        function showResults(data) {
            current = data.result;
            const o = OUTCOMES[current];
            const conf = Math.max(0, Math.min(100, data.confidence));

            $('verdict').textContent = o.label;
            $('verdict').style.color = o.color;
            $('verdictSub').textContent = o.sub;
            $('statusResult').textContent = o.label;
            $('statusResult').style.color = o.color;
            $('riskResult').textContent = o.risk;
            $('riskResult').style.color = o.color;
            $('fileResult').textContent = currentFile ? currentFile.name : '';
            $('specDesc').textContent = o.desc;
            $('tipsBox').style.setProperty('--c', o.color);
            $('tipsList').innerHTML = o.tips.map(t => '<li>' + t + '</li>').join('');

            $('resultsSection').style.display = 'block';
            drawSpectrogram($('spec'), current);

            const arc = $('arc'), C = 515.2;
            arc.style.stroke = o.color;
            arc.style.strokeDashoffset = C;
            requestAnimationFrame(() => requestAnimationFrame(() => {
                arc.style.strokeDashoffset = C * (1 - conf / 100);
            }));
            let n = 0;
            const timer = setInterval(() => {
                n = Math.min(n + 2.5, conf);
                $('pct').textContent = n.toFixed(1) + '%';
                if (n >= conf) clearInterval(timer);
            }, 20);

            $('resultsSection').scrollIntoView({ behavior: 'smooth', block: 'start' });
        }

        /* ===== رسم الـ Spectrogram (شكل توضيحي) ===== */
        function drawSpectrogram(cv, kind) {
            const x = cv.getContext('2d'), W = cv.width, H = cv.height;
            let seed = 7;
            const rnd = () => (seed = (seed * 1664525 + 1013904223) >>> 0) / 4294967296;
            x.fillStyle = '#070e1a'; x.fillRect(0, 0, W, H);
            for (let i = 0; i < W / 4; i++) {
                for (let j = 0; j < H / 6; j++) {
                    const f = j / (H / 6), n = rnd();
                    let v = Math.max(0, 0.9 - f * 1.1);
                    if (kind === 'fake') v = v * (0.55 + 0.25 * Math.sin(f * 40)) * (0.85 + 0.1 * Math.sin(i / 30)) + n * 0.04;
                    else if (kind === 'warn') v = v * (0.4 + 0.5 * n) * (i % 90 < 45 ? 1 : 0.75 + 0.2 * Math.sin(f * 40));
                    else v = v * (0.25 + 0.9 * n) * (0.6 + 0.5 * Math.pow(Math.sin(i / 9 + f * 7), 2));
                    v = Math.min(1, v);
                    x.fillStyle = 'hsl(' + (210 - v * 190) + ',85%,' + (12 + v * 55) + '%)';
                    x.fillRect(i * 4, H - j * 6 - 6, 4, 6);
                }
            }
        }

        /* ===== مشاركة + إعادة ===== */
        $('shareBtn').addEventListener('click', async () => {
            const o = OUTCOMES[current];
            const text = 'تقرير صقر (SAQR): ' + o.label + ' - نسبة الثقة ' + $('pct').textContent + ' - مستوى الخطر: ' + o.risk;
            try {
                if (navigator.share) await navigator.share({ text });
                else { await navigator.clipboard.writeText(text); $('shareBtn').textContent = 'تم النسخ ✓'; setTimeout(() => $('shareBtn').textContent = 'نسخ / مشاركة التقرير', 1500); }
            } catch (e) { /* المستخدم ألغى المشاركة */ }
        });

        $('againBtn').addEventListener('click', () => {
            fileInput.value = ''; currentFile = null;
            $('resultsSection').style.display = 'none';
            $('analyzeBtn').style.display = 'none';
            $('fileNameDisplay').textContent = 'اسحب وأفلت ملف الصوت هنا، أو اضغط لاختيار ملف';
            setError('');
            window.scrollTo({ top: 0, behavior: 'smooth' });
        });
    </script>
